import {
  Host,
  Text,
  TextField,
  type TextFieldRef,
  useNativeState,
} from "@expo/ui/swift-ui";

import {
  accessibilityLabel as accessibilityLabelModifier,
  autocorrectionDisabled,
  disabled as disabledModifier,
  foregroundStyle,
  keyboardType as keyboardTypeModifier,
  onSubmit,
  padding,
  submitLabel,
  textInputAutocapitalization,
  type ModifierConfig as IOSModifierConfig,
} from "@expo/ui/swift-ui/modifiers";

import { useCallback, useEffect, useRef } from "react";

import type { TextInputProps } from "./textinput.txt";

type IOSKeyboardType = Parameters<typeof keyboardTypeModifier>[0];

export default function TextInput({
  value,
  defaultValue = "",
  onChangeText,
  onSubmitEditing,
  onFocus,
  placeholder,
  placeholderTextColor = "#888888",
  textColor = "#FFFFFF",
  disabled = false,
  autoFocus = false,
  autoCorrect = true,
  keyboardType = "default",
  returnKeyType = "default",
  selectTextOnFocus = false,
  accessibilityLabel,
  autoCapitalize = "none",
  uiModifiers,
}: TextInputProps) {
  const inputRef = useRef<TextFieldRef>(null);
  const text = useNativeState(value ?? defaultValue);
  const selection = useNativeState({ start: 0, end: 0 });

  useEffect(() => {
    if (value !== undefined && text.value !== value) {
      text.value = value;
    }
  }, [value, text]);

  const handleTextChange = useCallback(
    (nextValue: string) => {
      onChangeText?.(nextValue);

      // Enforce controlled value if parent locks or transforms the value prop
      if (value !== undefined && nextValue !== value) {
        text.value = value;
      }
    },
    [onChangeText, value, text],
  );

  const handleFocus = useCallback(
    (focused: boolean) => {
      if (focused && selectTextOnFocus) {
        const len = text.value.length;
        selection.value = { start: 0, end: len };
        inputRef.current?.setSelection(0, len);
      }

      onFocus?.(focused);
    },
    [selectTextOnFocus, text, selection, onFocus],
  );

  const nativeKeyboardType: IOSKeyboardType =
    keyboardType === "visible-password"
      ? "default"
      : (keyboardType as IOSKeyboardType);

  const nativeSubmitLabel =
    returnKeyType === "go"
      ? "go"
      : returnKeyType === "search"
        ? "search"
        : returnKeyType === "send"
          ? "send"
          : returnKeyType === "next"
            ? "next"
            : returnKeyType === "done"
              ? "done"
              : "return";

  const modifiers: IOSModifierConfig[] = [
    foregroundStyle(textColor),
    padding({ vertical: 12 }),
    keyboardTypeModifier(nativeKeyboardType),
    ...(!autoCorrect ? [autocorrectionDisabled()] : []),
    textInputAutocapitalization(
      autoCapitalize === "none" ? "never" : autoCapitalize,
    ),
    submitLabel(nativeSubmitLabel),
    onSubmit(() => {
      onSubmitEditing?.({
        nativeEvent: { text: text.value },
      });
    }),
    ...(disabled ? [disabledModifier(true)] : []),
    ...(accessibilityLabel
      ? [accessibilityLabelModifier(accessibilityLabel)]
      : []),
    ...(uiModifiers?.ios ?? []),
  ];

  return (
    <Host matchContents={{ vertical: true }} style={{ width: "100%" }}>
      <TextField
        ref={inputRef}
        text={text}
        selection={selection}
        autoFocus={autoFocus}
        onTextChange={handleTextChange}
        onFocusChange={handleFocus}
        modifiers={modifiers}
      >
        {placeholder ? (
          <TextField.Placeholder>
            <Text modifiers={[foregroundStyle(placeholderTextColor)]}>
              {placeholder}
            </Text>
          </TextField.Placeholder>
        ) : undefined}
      </TextField>
    </Host>
  );
}
