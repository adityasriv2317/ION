import {
  Host,
  OutlinedTextField,
  type TextFieldRef,
  Text,
  useNativeState,
} from "@expo/ui/jetpack-compose";

import {
  fillMaxWidth,
  type ModifierConfig as AndroidModifierConfig,
} from "@expo/ui/jetpack-compose/modifiers";

import type { ModifierConfig as IOSModifierConfig } from "@expo/ui/swift-ui/modifiers";

import { useCallback, useEffect, useRef } from "react";

export type KeyboardType =
  | "default"
  | "email-address"
  | "numeric"
  | "phone-pad"
  | "decimal-pad"
  | "url"
  | "web-search"
  | "ascii-capable"
  | "visible-password";

export type ReturnKeyType =
  | "default"
  | "go"
  | "search"
  | "send"
  | "next"
  | "done";

export type TextInputProps = {
  value?: string;
  defaultValue?: string;
  onChangeText?: (value: string) => void;
  onSubmitEditing?: (event: { nativeEvent: { text: string } }) => void;
  onFocus?: (focused: boolean) => void;

  placeholder?: string;
  keyboardType?: KeyboardType;
  returnKeyType?: ReturnKeyType;
  
  placeholderTextColor?: string;
  textColor?: string;
  disabled?: boolean;
  autoFocus?: boolean;
  autoCorrect?: boolean;
  selectTextOnFocus?: boolean;
  accessibilityLabel?: string;
  autoCapitalize?: "none" | "sentences" | "words" | "characters";

  uiModifiers?: {
    ios?: IOSModifierConfig[];
    android?: AndroidModifierConfig[];
  };
};

export default function TextInput({
  value,
  defaultValue = "",
  onChangeText,
  onSubmitEditing,
  onFocus,
  placeholder,
  placeholderTextColor = "#888888",
  disabled = false,
  autoFocus = false,
  autoCorrect = true,
  keyboardType = "default",
  returnKeyType = "default",
  selectTextOnFocus = false,
  accessibilityLabel,
  autoCapitalize = "none",
  uiModifiers,
}: TextInputProps) {
  const inputRef = useRef<TextFieldRef>(null);
  const text = useNativeState(value ?? defaultValue);
  const selection = useNativeState({ start: 0, end: 0 });

  useEffect(() => {
    if (value !== undefined && text.value !== value) {
      text.value = value;
    }
  }, [value, text]);

  const handleValueChange = useCallback(
    (nextValue: string) => {
      onChangeText?.(nextValue);

      // Enforce controlled value if parent locks or transforms the value prop
      if (value !== undefined && nextValue !== value) {
        text.value = value;
      }
    },
    [onChangeText, value, text],
  );

  const handleSubmit = useCallback(
    (submittedText: string) => {
      onSubmitEditing?.({
        nativeEvent: { text: submittedText },
      });
    },
    [onSubmitEditing],
  );

  const handleFocus = useCallback(
    (focused: boolean) => {
      if (focused && selectTextOnFocus) {
        const len = text.value.length;
        selection.value = { start: 0, end: len };
        inputRef.current?.setSelection(0, len);
      }

      onFocus?.(focused);
    },
    [selectTextOnFocus, text, selection, onFocus],
  );

  const imeAction =
    returnKeyType === "go"
      ? "go"
      : returnKeyType === "search"
        ? "search"
        : returnKeyType === "send"
          ? "send"
          : returnKeyType === "next"
            ? "next"
            : returnKeyType === "done"
              ? "done"
              : "default";

  const nativeKeyboardType =
    keyboardType === "web-search" || keyboardType === "url"
      ? "uri"
      : keyboardType === "email-address"
        ? "email"
        : keyboardType === "numeric"
          ? "number"
          : keyboardType === "phone-pad"
            ? "phone"
            : keyboardType === "decimal-pad"
              ? "decimal"
              : keyboardType === "visible-password"
                ? "password"
                : keyboardType === "ascii-capable"
                  ? "ascii"
                  : "text";

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: "100%" }}
      // accessibilityLabel={accessibilityLabel}
    >
      <OutlinedTextField
        ref={inputRef}
        value={text}
        selection={selection}
        enabled={!disabled}
        autoFocus={autoFocus}
        singleLine
        onValueChange={handleValueChange}
        onFocusChanged={handleFocus}
        keyboardOptions={{
          keyboardType: nativeKeyboardType,
          imeAction,
          autoCorrectEnabled: autoCorrect,
          capitalization:
            autoCapitalize === "characters"
              ? "characters"
              : autoCapitalize === "words"
                ? "words"
                : autoCapitalize === "sentences"
                  ? "sentences"
                  : "none",
        }}
        keyboardActions={{
          onGo: handleSubmit,
          onSearch: handleSubmit,
          onSend: handleSubmit,
          onNext: handleSubmit,
          onDone: handleSubmit,
        }}
        modifiers={[fillMaxWidth(), ...(uiModifiers?.android ?? [])]}
      >
        {placeholder ? (
          <OutlinedTextField.Placeholder>
            <Text color={placeholderTextColor}>{placeholder}</Text>
          </OutlinedTextField.Placeholder>
        ) : null}
      </OutlinedTextField>
    </Host>
  );
}
